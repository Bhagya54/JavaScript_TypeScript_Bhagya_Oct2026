📜 Normal

validateStatusCode(203); // works!
function validateStatusCode(status) {
  if (status >= 200 && status < 300) {
    console.log("Request is fine:", status)
  }
}

📦 Expression

const validateStatusCode_Exp =
  function (status) {
    if (status >= 200 && status < 300) {
      console.log("Request is fine:", status)
    }
  }
validateStatusCode_Exp(205);

➡️ Arrow

const validateStatusCode_Arrow =
  (status) => {
    if (status >= 200 && status < 300) {
      console.log("Request is fine:", status)
    }
  }
validateStatusCode_Arrow(201)

🧠 All three did the same job in our file. Difference = when you can call them and how this behaves.
